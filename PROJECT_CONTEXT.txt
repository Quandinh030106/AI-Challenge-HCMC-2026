import numpy as np

class Evaluator:
    def __init__(self, ground_truth):
        """
        ground_truth: Dict Ä‘á»c tá»« file local_val_gt.json
        """
        # Chuyá»ƒn Ä‘á»•i danh sÃ¡ch Ground Truth thÃ nh tá»« Ä‘iá»ƒn tra cá»©u nhanh theo query_id
        self.gt = {}
        for task in ["task1", "task2", "task3"]:
            self.gt[task] = {}
            if task in ground_truth:
                # Há»— trá»£ cáº£ trÆ°á»ng há»£p Ground Truth lÃ  List hoáº·c Dict
                if isinstance(ground_truth[task], list):
                    for item in ground_truth[task]:
                        self.gt[task][item["query_id"]] = item
                elif isinstance(ground_truth[task], dict):
                    self.gt[task] = ground_truth[task]
        
    def check_answer_match(self, pred, gt):
        """
        Kiá»ƒm tra Ä‘á»™ trÃ¹ng khá»›p cÃ¢u tráº£ lá»i Q&A má»™t cÃ¡ch má»m dáº»o (Normalized Semantic Match).
        """
        if not pred or not gt:
            return False
        pred = str(pred).lower().strip()
        gt = str(gt).lower().strip()
        
        # Khá»›p tuyá»‡t Ä‘á»‘i
        if pred == gt:
            return True
        # Khá»›p chuá»—i con
        if gt in pred or pred in gt:
            return True
        # Khá»›p cÃ¡c tá»« khÃ³a cá»‘t lÃµi (Giao cá»§a táº­p há»£p tá»«)
        pred_words = set(pred.split())
        gt_words = set(gt.split())
        if len(pred_words.intersection(gt_words)) > 0:
            return True
        return False

    def evaluate_query(self, query_id, task_type, predictions):
        """
        ÄÃ¡nh giÃ¡ 1 cÃ¢u query duy nháº¥t.
        predictions: Danh sÃ¡ch tá»‘i Ä‘a 100 cÃ¢u tráº£ lá»i ná»™p lÃªn, sáº¯p xáº¿p thá»© tá»± Æ°u tiÃªn giáº£m dáº§n.
        """
        if query_id not in self.gt.get(task_type, {}):
            return 0.0, [0.0] * 5
            
        gt_item = self.gt[task_type][query_id]
        gt_video = gt_item["video_id"]
        
        # 1. TÃ­nh R-Score cho tá»«ng dá»± Ä‘oÃ¡n thá»© i trong danh sÃ¡ch ná»™p (tá»‘i Ä‘a 100)
        r_scores = []
        for pred in predictions:
            pred_video = pred.get("video_id")
            
            # Sai video nháº­n ngay 0 Ä‘iá»ƒm
            if pred_video != gt_video:
                r_scores.append(0.0)
                continue
                
            if task_type == "task1":
                # KIS: ÄÃºng video vÃ  frame náº±m trong khoáº£ng cho phÃ©p [s, e]
                try:
                    pred_frame = int(pred.get("frame_id", -1))
                    frame_start = int(gt_item["frame_start"])
                    frame_end = int(gt_item["frame_end"])
                    if frame_start <= pred_frame <= frame_end:
                        r_scores.append(1.0)
                    else:
                        r_scores.append(0.0)
                except ValueError:
                    r_scores.append(0.0)
                    
            elif task_type == "task2":
                # Q&A: ÄÃºng video, Ä‘Ãºng frame vÃ  Ä‘Ãºng cÃ¢u tráº£ lá»i phá»¥
                try:
                    pred_frame = int(pred.get("frame_id", -1))
                    frame_start = int(gt_item["frame_start"])
                    frame_end = int(gt_item["frame_end"])
                    pred_ans = pred.get("answer", "")
                    gt_ans = gt_item["answer"]
                    
                    if (frame_start <= pred_frame <= frame_end) and self.check_answer_match(pred_ans, gt_ans):
                        r_scores.append(1.0)
                    else:
                        r_scores.append(0.0)
                except ValueError:
                    r_scores.append(0.0)
                    
            elif task_type == "task3":
                # TRAKE: Tá»· lá»‡ khá»›p khung hÃ¬nh cá»§a chuá»—i sá»± kiá»‡n con
                events = gt_item["events"]
                pred_frames = pred.get("frame_ids", [])
                
                if len(pred_frames) != len(events):
                    r_scores.append(0.0)
                    continue
                    
                match_count = 0
                for idx, event in enumerate(events):
                    try:
                        pf = int(pred_frames[idx])
                        fs = int(event["frame_start"])
                        fe = int(event["frame_end"])
                        if fs <= pf <= fe:
                            match_count += 1
                    except ValueError:
                        pass
                # R-Score = tá»· lá»‡ sá»‘ event khá»›p trÃªn tá»•ng sá»‘ event
                r_scores.append(match_count / len(events))
                
        # Äiá»n thÃªm Ä‘iá»ƒm 0.0 cho Ä‘á»§ 100 cÃ¢u tráº£ lá»i
        if not r_scores:
            r_scores = [0.0]
        while len(r_scores) < 100:
            r_scores.append(0.0)
            
        # 2. TÃ­nh R@k vá»›i k thuá»™c {1, 5, 20, 50, 100}
        k_values = [1, 5, 20, 50, 100]
        r_at_k = []
        for k in k_values:
            val = float(max(r_scores[:k]))
            r_at_k.append(val)
            
        # 3. Äiá»ƒm Final Score cá»§a query lÃ  trung bÃ¬nh cá»™ng cá»§a 5 má»‘c R@k
        final_score = float(np.mean(r_at_k))
        return final_score, r_at_k

    def evaluate_all(self, predictions_dict):
        """
        TÃ­nh Ä‘iá»ƒm trung bÃ¬nh toÃ n bá»™ táº¥t cáº£ cÃ¡c Task.
        predictions_dict: Káº¿t quáº£ cháº¡y cá»§a cáº£ nhÃ³m:
        {
          "task1": { "query_id_1": [pred1, pred2, ...], ... },
          "task2": { ... },
          "task3": { ... }
        }
        """
        all_final_scores = []
        r_at_k_accum = np.zeros(5)
        count = 0
        
        for task_type in ["task1", "task2", "task3"]:
            task_gt = self.gt.get(task_type, {})
            task_preds = predictions_dict.get(task_type, {})
            
            for query_id in task_gt.keys():
                preds = task_preds.get(query_id, [])
                final_score, r_at_k = self.evaluate_query(query_id, task_type, preds)
                all_final_scores.append(final_score)
                r_at_k_accum += np.array(r_at_k)
                count += 1
                
        if count == 0:
            return {"Final_Score": 0.0, "R@1": 0.0, "R@5": 0.0, "R@20": 0.0, "R@50": 0.0, "R@100": 0.0}
            
        r_at_k_avg = r_at_k_accum / count
        mean_final_score = float(np.mean(all_final_scores))
        
        return {
            "Final_Score": mean_final_score,
            "R@1": float(r_at_k_avg[0]),
            "R@5": float(r_at_k_avg[1]),
            "R@20": float(r_at_k_avg[2]),
            "R@50": float(r_at_k_avg[3]),
            "R@100": float(r_at_k_avg[4])
        }

import os
import glob
import numpy as np
import torch
from PIL import Image
from tqdm import tqdm
from transformers import AutoProcessor, AutoModel
from src.utils import load_config

def extract_all_features(config_path="configs/default.yaml", batch_size=32):
    # 1. Load cáº¥u hÃ¬nh Ä‘Æ°á»ng dáº«n
    config = load_config(config_path)
    keyframes_dir = config["data"]["keyframes_dir"]
    features_dir = config["data"]["features_dir"]
    model_name = config["models"]["clip_model"]
    
    os.makedirs(features_dir, exist_ok=True)
    
    # 2. Khá»Ÿi táº¡o mÃ´ hÃ¬nh chuyÃªn biá»‡t
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"Äang khá»Ÿi táº¡o model: {model_name} trÃªn thiáº¿t bá»‹: {device}...")
    
    from transformers import (
        CLIPModel, CLIPProcessor, 
        SiglipModel, SiglipProcessor, 
        AutoModel, AutoProcessor
    )
    
    if "siglip" in model_name.lower():
        processor = SiglipProcessor.from_pretrained(model_name)
        model = SiglipModel.from_pretrained(model_name).to(device)
    elif "clip" in model_name.lower():
        processor = CLIPProcessor.from_pretrained(model_name)
        model = CLIPModel.from_pretrained(model_name).to(device)
    else:
        processor = AutoProcessor.from_pretrained(model_name)
        model = AutoModel.from_pretrained(model_name).to(device)
        
    model.eval()

    
    # 3. Láº¥y táº¥t cáº£ cÃ¡c thÆ° má»¥c chá»©a keyframe (má»—i thÆ° má»¥c tÆ°Æ¡ng á»©ng vá»›i má»™t video_id)
    video_dirs = [d for d in glob.glob(os.path.join(keyframes_dir, "*")) if os.path.isdir(d)]
    print(f"TÃ¬m tháº¥y {len(video_dirs)} thÆ° má»¥c keyframe video cáº§n xá»­ lÃ½.")
    
    for video_dir in tqdm(video_dirs, desc="TrÃ­ch xuáº¥t Ä‘áº·c trÆ°ng"):
        video_id = os.path.basename(video_dir)
        output_path = os.path.join(features_dir, f"{video_id}.npy")
        
        # Bá» qua náº¿u Ä‘Ã£ trÃ­ch xuáº¥t trÆ°á»›c Ä‘Ã³
        if os.path.exists(output_path):
            continue
            
        # Láº¥y vÃ  sáº¯p xáº¿p thá»© tá»± cÃ¡c keyframe
        image_paths = sorted(glob.glob(os.path.join(video_dir, "*.jpg")))
        if not image_paths:
            print(f"Cáº£nh bÃ¡o: KhÃ´ng tÃ¬m tháº¥y áº£nh trong thÆ° má»¥c {video_dir}")
            continue
            
        video_features = []
        
        # 4. Xá»­ lÃ½ áº£nh theo tá»«ng Batch Ä‘á»ƒ tá»‘i Æ°u tÃ i nguyÃªn GPU
        for i in range(0, len(image_paths), batch_size):
            batch_paths = image_paths[i:i+batch_size]
            batch_images = []
            
            for img_path in batch_paths:
                try:
                    img = Image.open(img_path).convert("RGB")
                    batch_images.append(img)
                except Exception as e:
                    print(f"Lá»—i Ä‘á»c áº£nh {img_path}: {e}")
                    
            if not batch_images:
                continue
                
            # ÄÆ°a qua bá»™ tiá»n xá»­ lÃ½ vÃ  trÃ­ch xuáº¥t vector Ä‘áº·c trÆ°ng
            inputs = processor(images=batch_images, return_tensors="pt").to(device)
            with torch.no_grad():
                outputs = model.get_image_features(**inputs)
                # Chuáº©n hÃ³a L2-norm Ä‘á»ƒ cÃ³ thá»ƒ tÃ­nh Cosine Similarity trá»±c tiáº¿p báº±ng Dot Product
                outputs = outputs / outputs.norm(p=2, dim=-1, keepdim=True)
                features = outputs.cpu().numpy()
                video_features.append(features)
                
        if video_features:
            # GhÃ©p ná»‘i cÃ¡c batch láº¡i thÃ nh 1 tensor dáº¡ng [num_frames, feature_dim]
            video_features = np.concatenate(video_features, axis=0)
            np.save(output_path, video_features)
            
    print("QuÃ¡ trÃ¬nh trÃ­ch xuáº¥t Ä‘áº·c trÆ°ng áº£nh SigLIP hoÃ n thÃ nh!")

if __name__ == "__main__":
    extract_all_features()

            print("QueryProcessor: âœ… ÄÃ£ khá»Ÿi táº¡o thÃ nh cÃ´ng bá»™ dá»‹ch Tiáº¿ng Viá»‡t -> Tiáº¿ng Anh offline.")
        except Exception as e:
            print(f"QueryProcessor: Cáº£nh bÃ¡o khÃ´ng náº¡p Ä‘Æ°á»£c Opus-MT ({e}), sá»­ dá»¥ng cháº¿ Ä‘á»™ fallback giá»¯ nguyÃªn query.")
            self.translator_available = False

    def translate_vi_to_en(self, text_vi):
        """Dá»‹ch cÃ¢u truy váº¥n Tiáº¿ng Viá»‡t sang Tiáº¿ng Anh Ä‘á»ƒ tá»‘i Æ°u mÃ´ hÃ¬nh CLIP/SigLIP."""
        if not self.translator_available or not text_vi.strip():
            return text_vi
            
        try:
            inputs = self.tokenizer(text_vi, return_tensors="pt", padding=True).to(self.device)
            with torch.no_grad():
                translated_tokens = self.model.generate(**inputs, max_length=100)
            translated_text = self.tokenizer.batch_decode(translated_tokens, skip_special_tokens=True)[0]
            return translated_text
        except Exception as e:
            return text_vi

    def detect_query_intent(self, text_vi):
        """
        Nháº­n diá»‡n Ã½ Ä‘á»‹nh cÃ¢u há»i Ä‘á»ƒ gÃ¡n trá»ng sá»‘ RRF Fusion:
        - Náº¿u chá»©a tá»« khÃ³a nhÆ° 'biá»ƒn bÃ¡o', 'chá»¯', 'báº£ng hiá»‡u', 'sá»‘ xe', 'viáº¿t tá»«' -> Æ¯u tiÃªn BM25 OCR.
        - NgÆ°á»£c láº¡i -> Æ¯u tiÃªn Vector Search (CLIP/SigLIP).
        """
        text_lower = text_vi.lower()
        ocr_keywords = [
            "chá»¯", "biá»ƒn bÃ¡o", "báº£ng hiá»‡u", "tÃªn", "sá»‘ xe", "biá»ƒn sá»‘", 
            "dÃ²ng chá»¯", "viáº¿t", "kÃ½ tá»±", "báº£ng tÃªn", "Ã¡o ghi sá»‘", "logo"
        ]
        
        for kw in ocr_keywords:
            if kw in text_lower:
                return {
                    "intent": "OCR_TEXT",
                    "dense_weight": 0.3,
                    "sparse_weight": 0.7
                }
                
        return {
            "intent": "VISUAL_SCENE",
            "dense_weight": 0.7,
            "sparse_weight": 0.3
        }

    def process(self, query_vi):
        """
        Xá»­ lÃ½ toÃ n diá»‡n cÃ¢u truy váº¥n Ä‘áº§u vÃ o.
        Tráº£ vá» dictionary chá»©a:

        - intent_info: ThÃ´ng tin Ã½ Ä‘á»‹nh vÃ  trá»ng sá»‘ gá»™p Ä‘iá»ƒm
        """
        query_en = self.translate_vi_to_en(query_vi)
        intent_info = self.detect_query_intent(query_vi)
        
        return {
            "query_vi": query_vi,
            "query_en": query_en,
            "intent_info": intent_info
        }

def run_ocr_on_all_keyframes(config_path="configs/default.yaml", step=5):

    # 2. Khá»Ÿi Ä‘á»™ng EasyOCR Reader (há»— trá»£ Tiáº¿ng Viá»‡t 'vi' vÃ  Tiáº¿ng Anh 'en')
    use_gpu = torch.cuda.is_available()
    print(f"Khá»Ÿi táº¡o EasyOCR (vi + en) trÃªn thiáº¿t bá»‹ GPU={use_gpu}...")
    reader = easyocr.Reader(['vi', 'en'], gpu=use_gpu)
    
    # 3. QuÃ©t táº¥t cáº£ cÃ¡c thÆ° má»¥c chá»©a keyframe
    video_dirs = [d for d in glob.glob(os.path.join(keyframes_dir, "*")) if os.path.isdir(d)]
    print(f"TÃ¬m tháº¥y {len(video_dirs)} thÆ° má»¥c video Ä‘á»ƒ cháº¡y OCR.")
    
    for video_dir in tqdm(video_dirs, desc="Cháº¡y OCR trÃªn video"):
        video_id = os.path.basename(video_dir)
        output_path = os.path.join(metadata_dir, f"{video_id}_ocr.json")
        
        # Bá» qua náº¿u video nÃ y Ä‘Ã£ Ä‘Æ°á»£c OCR trÆ°á»›c Ä‘Ã³
        if os.path.exists(output_path):

        # 5. LÆ°u káº¿t quáº£ OCR cá»§a video
        with open(output_path, "w", encoding="utf-8") as f:

    print("QuÃ¡ trÃ¬nh cháº¡y OCR trÃªn toÃ n bá»™ keyframes hoÃ n thÃ nh!")

import os
import glob
import numpy as np
import torch
from transformers import (
    CLIPModel, CLIPProcessor, 
    SiglipModel, SiglipProcessor, 
    AutoModel, AutoProcessor
)

class DenseSearcher:
    def __init__(self, config):
        self.config = config
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.model_name = config["models"]["clip_model"]
        self.features_dir = config["data"]["features_dir"]
        
        print(f"DenseSearcher: Äang khá»Ÿi táº¡o mÃ´ hÃ¬nh {self.model_name} trÃªn {self.device}...")
        
        # 1. Khá»Ÿi táº¡o mÃ´ hÃ¬nh mÃ£ hÃ³a text
        if "siglip" in self.model_name.lower():
            self.processor = SiglipProcessor.from_pretrained(self.model_name)
            self.model = SiglipModel.from_pretrained(self.model_name).to(self.device)
        elif "clip" in self.model_name.lower():
            self.processor = CLIPProcessor.from_pretrained(self.model_name)
            self.model = CLIPModel.from_pretrained(self.model_name).to(self.device)
        else:
            self.processor = AutoProcessor.from_pretrained(self.model_name)
            self.model = AutoModel.from_pretrained(self.model_name).to(self.device)
            
        self.model.eval()
        
        # 2. Bá»™ nhá»› Ä‘á»‡m ma tráº­n toÃ n cá»¥c (Global Tensor Cache) Ä‘á»ƒ Ä‘áº¡t Ä‘á»™ trá»… < 5ms
        self.global_tensor = None
        self.video_metadata_list = [] # LÆ°u danh sÃ¡ch (video_id, total_frames, start_idx, end_idx)
        self.all_video_ids = []
        self.video_features_dict = {} # LÆ°u thÃ´ Ä‘á»ƒ Task 1 / Task 3 dÃ¹ng láº¡i
        
        self.load_and_build_global_matrix()

    def load_and_build_global_matrix(self):
        """QuÃ©t vÃ  gá»™p táº¥t cáº£ cÃ¡c file .npy thÃ nh 1 Ma tráº­n GPU duy nháº¥t siÃªu tá»‘c."""
        print(f"DenseSearcher: Äang náº¡p toÃ n bá»™ vector Ä‘áº·c trÆ°ng tá»« thÆ° má»¥c: {self.features_dir}")
        
        feature_files = []
        
        # 1. QuÃ©t trong /kaggle/input báº¥t ká»³ file nÃ o cÃ³ Ä‘uÃ´i .npy (bá» qua thÆ° má»¥c keyframes vÃ  videos Ä‘á»ƒ siÃªu nhanh)
        if os.path.exists("/kaggle/input"):
            for root, _, files in os.walk("/kaggle/input"):
                root_lower = root.lower()
                if "keyframe" in root_lower or "video" in root_lower:
                    continue # Nháº£y qua ná»­a triá»‡u file áº£nh/video Ä‘á»ƒ cháº¡y trong 0.05 giÃ¢y
                for file in files:
                    if file.lower().endswith(".npy"):
                        feature_files.append(os.path.join(root, file))

        print(f"DenseSearcher: ÄÃ£ tÃ¬m tháº¥y thÃ nh cÃ´ng {len(feature_files)} file .npy Ä‘áº·c trÆ°ng.")
        if feature_files:
            print(f"DenseSearcher: VÃ­ dá»¥ 3 file .npy tÃ¬m tháº¥y: {[os.path.basename(f) for f in feature_files[:3]]}")
        else:
            print("Cáº£nh bÃ¡o: KhÃ´ng tÃ¬m tháº¥y file .npy nÃ o trong /kaggle/input!")
            return



        all_vectors = []
        current_idx = 0
        
        for file_path in feature_files:
            video_id = os.path.splitext(os.path.basename(file_path))[0]
            try:
                feats = np.load(file_path) # Shape: (N_frames, Dim)
                # Chuáº©n hÃ³a L2 cho cÃ¡c vector áº£nh náº¿u chÆ°a chuáº©n hÃ³a
                norms = np.linalg.norm(feats, axis=-1, keepdims=True)
                norms[norms == 0] = 1e-10
                feats_norm = feats / norms
                
                n_frames = feats_norm.shape[0]
                self.video_features_dict[video_id] = feats_norm
                
                all_vectors.append(feats_norm)
                self.video_metadata_list.append({
                    "video_id": video_id,
                    "start_idx": current_idx,
                    "end_idx": current_idx + n_frames,
                    "n_frames": n_frames
                })
                self.all_video_ids.append(video_id)
                current_idx += n_frames
            except Exception as e:
                pass
                
        if all_vectors:
            concat_matrix = np.vstack(all_vectors) # Shape: (Total_Frames_All_Videos, Dim)
            self.global_tensor = torch.from_numpy(concat_matrix).float().to(self.device)
            print(f"DenseSearcher: âœ… ÄÃ£ náº¡p thÃ nh cÃ´ng Ma tráº­n toÃ n cá»¥c {self.global_tensor.shape} ({len(self.all_video_ids)} videos) vÃ o {self.device}. Äá»™ trá»… tÃ¬m kiáº¿m < 5ms!")

    def encode_text(self, text):
        """MÃ£ hÃ³a vÄƒn báº£n truy váº¥n thÃ nh vector Ä‘áº·c trÆ°ng vÃ  chuáº©n hÃ³a L2."""
        inputs = self.processor(text=[text], padding=True, return_tensors="pt").to(self.device)
        with torch.no_grad():
            text_outputs = self.model.get_text_features(**inputs)
            if isinstance(text_outputs, torch.Tensor):
                text_features = text_outputs
            elif hasattr(text_outputs, "text_embeds"):
                text_features = text_outputs.text_embeds
            elif hasattr(text_outputs, "pooler_output"):
                text_features = text_outputs.pooler_output
            else:
                text_features = text_outputs[0]
                
            text_features = text_features / text_features.norm(p=2, dim=-1, keepdim=True)
            return text_features

    def search(self, query_text, top_k_videos=10):
        """TÃ¬m kiáº¿m siÃªu tá»‘c trÃªn Ma tráº­n GPU toÃ n cá»¥c báº±ng phÃ©p nhÃ¢n ma tráº­n (Matrix Multiplication)."""
        if self.global_tensor is None:
            print("Cáº£nh bÃ¡o: Ma tráº­n Ä‘áº·c trÆ°ng chÆ°a Ä‘Æ°á»£c khá»Ÿi táº¡o!")
            return []

        # 1. MÃ£ hÃ³a query (Shape: 1 x Dim)
        q_tensor = self.encode_text(query_text)
        
        # 2. NhÃ¢n ma tráº­n siÃªu tá»‘c trÃªn GPU: (1 x Dim) x (Dim x Total_Frames) -> (Total_Frames,)
        with torch.no_grad():
            sim_scores = torch.matmul(self.global_tensor, q_tensor.T).squeeze(-1) # Shape: (Total_Frames,)
            sim_scores_np = sim_scores.cpu().numpy()

        # 3. Tá»•ng há»£p Ä‘iá»ƒm max/mean cho tá»«ng video
        results = []
        for meta in self.video_metadata_list:
            video_id = meta["video_id"]
            start_i = meta["start_idx"]
            end_i = meta["end_idx"]
            
            v_scores = sim_scores_np[start_i:end_i]
            max_idx = np.argmax(v_scores)
            max_score = float(v_scores[max_idx])
            
            results.append({
                "video_id": video_id,
                "max_score": max_score,
                "best_frame_idx": int(max_idx),
                "all_scores": v_scores
            })

        # 4. Sáº¯p xáº¿p giáº£m dáº§n theo max_score
        results.sort(key=lambda x: x["max_score"], reverse=True)
        return results[:top_k_videos]

def reciprocal_rank_fusion(dense_results, sparse_results, k=60, dense_weight=1.0, sparse_weight=1.0):
    """
    Káº¿t há»£p káº¿t quáº£ tá»« Dense Search (CLIP/SigLIP) vÃ  Sparse Search (BM25)
    sá»­ dá»¥ng thuáº­t toÃ¡n Reciprocal Rank Fusion (RRF) cÃ³ trá»ng sá»‘ Ä‘á»™ng.
    """
    rrf_scores = {}
    
    # 1. TÃ­nh toÃ¡n thá»© háº¡ng vÃ  Ä‘iá»ƒm RRF cho Dense Search
    for rank, item in enumerate(dense_results):
        video_id = item["video_id"]
        if video_id not in rrf_scores:
            rrf_scores[video_id] = {
                "video_id": video_id,
                "rrf_score": 0.0,
                "dense_info": item
            }
        rrf_scores[video_id]["rrf_score"] += dense_weight * (1.0 / (k + rank + 1))
        
    # 2. TÃ­nh toÃ¡n thá»© háº¡ng vÃ  Ä‘iá»ƒm RRF cho Sparse Search
    for rank, item in enumerate(sparse_results):
        video_id = item["video_id"]
        if video_id not in rrf_scores:
            rrf_scores[video_id] = {
                "video_id": video_id,
                "rrf_score": 0.0,
                "dense_info": None
            }
        rrf_scores[video_id]["rrf_score"] += sparse_weight * (1.0 / (k + rank + 1))
        
    # 3. Sáº¯p xáº¿p láº¡i danh sÃ¡ch káº¿t quáº£ theo Ä‘iá»ƒm RRF giáº£m dáº§n
    fused_results = list(rrf_scores.values())
    fused_results = sorted(fused_results, key=lambda x: x["rrf_score"], reverse=True)
    return fused_results

        if not text:

    def extract_text_from_obj(self, obj):
        """BÃ³c tÃ¡ch Ä‘á»‡ quy táº¥t cáº£ cÃ¡c chuá»—i vÄƒn báº£n trong file JSON (ká»ƒ cáº£ trong List hay Dict)."""
        text = ""

            text += " " + obj
        elif isinstance(obj, list):
            for item in obj:
                text += self.extract_text_from_obj(item)
        elif isinstance(obj, dict):
            for val in obj.values():
                text += self.extract_text_from_obj(val)
        return text

    def build_index(self):

                if video_id not in video_texts:

def solve_task1(query_text, fused_candidates, keyframes_dir, window_size=5):
    """
    Giáº£i quyáº¿t Task 1: TÃ¬m kiáº¿m chÃ­nh xÃ¡c (KIS)
    1. Nháº­n video á»©ng viÃªn hÃ ng Ä‘áº§u tá»« káº¿t quáº£ Fusion.
    2. Ãp dá»¥ng bá»™ lá»c lÃ m má»‹n (Rolling Average) trÃªn cÃ¡c frame Ä‘á»ƒ khá»­ nhiá»…u.
    3. Tráº£ vá» video_id vÃ  frame_id cá»§a Ä‘iá»ƒm tÆ°Æ¡ng Ä‘á»“ng cao nháº¥t.
    """
    if not fused_candidates:
        return {"video_id": "none", "frame_id": "0000", "score": 0.0}
        
    # Chá»n á»©ng viÃªn tá»‘t nháº¥t tá»« káº¿t quáº£ Fusion
    best_candidate = fused_candidates[0]
    video_id = best_candidate["video_id"]
    dense_info = best_candidate["dense_info"]
    
    if dense_info is None or "all_scores" not in dense_info:
        # Náº¿u chá»‰ tÃ¬m tháº¥y qua BM25 khÃ´ng cÃ³ thÃ´ng tin vector Ä‘áº·c trÆ°ng
        return {
            "video_id": video_id,
            "frame_id": "0000",
            "score": best_candidate.get("rrf_score", 0.0)
        }
        
    scores = dense_info["all_scores"]
    
    # Ãp dá»¥ng bá»™ lá»c lÃ m má»‹n (Moving Average)
    if len(scores) >= window_size:
        smoothed_scores = np.convolve(scores, np.ones(window_size) / window_size, mode='same')
    else:
        smoothed_scores = scores
        
    # TÃ¬m chá»‰ sá»‘ frame cÃ³ Ä‘iá»ƒm cao nháº¥t sau khi lá»c
    best_frame_idx = int(np.argmax(smoothed_scores))
    best_score = float(smoothed_scores[best_frame_idx])
    
    # Ãnh xáº¡ chá»‰ sá»‘ sang frame_id thá»±c táº¿ (vÃ­ dá»¥: '0150')
    frame_id = get_frame_id_from_idx(keyframes_dir, video_id, best_frame_idx)
    
    return {
        "video_id": video_id,
        "frame_id": frame_id,
        "score": best_score
    }

import os
import glob
import numpy as np
from src.tasks.task1_kis import get_frame_id_from_idx

def align_events_dynamic_programming(scores_matrix):
    """
    Sá»­ dá»¥ng Quy hoáº¡ch Ä‘á»™ng (Dynamic Programming) Ä‘á»ƒ tÃ¬m chuá»—i frame t_1 < t_2 < ... < t_N
    sao cho tá»•ng Ä‘iá»ƒm tÆ°Æ¡ng Ä‘á»“ng cá»§a chuá»—i sá»± kiá»‡n Ä‘Æ°á»£c cá»±c Ä‘áº¡i hÃ³a.
    
    scores_matrix: Ma tráº­n cÃ³ kÃ­ch thÆ°á»›c [T, N], chá»©a Ä‘iá»ƒm cosine similarity
                   giá»¯a T frame áº£nh vÃ  N cÃ¢u mÃ´ táº£ sá»± kiá»‡n con.
    """
    T, N = scores_matrix.shape
    if T < N:
        # Náº¿u sá»‘ frame Ã­t hÆ¡n sá»‘ sá»± kiá»‡n, khÃ´ng thá»ƒ sáº¯p xáº¿p tuyáº¿n tÃ­nh
        return list(range(T)) + [T - 1] * (N - T), 0.0

    # dp[t, j] lÃ  Ä‘iá»ƒm sá»‘ cá»±c Ä‘áº¡i khi cÄƒn chá»‰nh j sá»± kiá»‡n Ä‘áº§u tiÃªn vÃ o cÃ¡c frame tá»« 0 Ä‘áº¿n t
    dp = np.full((T, N + 1), -np.inf)
    # parent dÃ¹ng Ä‘á»ƒ lÆ°u váº¿t Ä‘Æ°á»ng Ä‘i phá»¥c vá»¥ backtracking
    parent = np.full((T, N + 1), -1)

    # Khá»Ÿi táº¡o: CÄƒn chá»‰nh 0 sá»± kiá»‡n cÃ³ Ä‘iá»ƒm lÃ  0
    for t in range(T):
        dp[t, 0] = 0.0

    for j in range(1, N + 1):
        for t in range(j - 1, T):
            # Lá»±a chá»n 1: KhÃ´ng chá»n frame t cho sá»± kiá»‡n j (sá»± kiá»‡n j Ä‘Ã£ Ä‘Æ°á»£c cÄƒn chá»‰nh vÃ o frame trÆ°á»›c t)
            if t > 0:
                dp[t, j] = dp[t - 1, j]
                parent[t, j] = t - 1

            # Lá»±a chá»n 2: Chá»n frame t cho sá»± kiá»‡n j (sá»± kiá»‡n j-1 pháº£i Ä‘Æ°á»£c chá»n trÆ°á»›c frame t, tá»©c lÃ  tá»« frame 0 Ä‘áº¿n t-1)
            prev_score = dp[t - 1, j - 1] if t > 0 else (0.0 if j == 1 else -np.inf)
            current_score = prev_score + scores_matrix[t, j - 1]

            if current_score > dp[t, j]:
                dp[t, j] = current_score
                parent[t, j] = -2  # ÄÃ¡nh dáº¥u chuyá»ƒn tráº¡ng thÃ¡i chá»n frame t cho sá»± kiá»‡n j

    # Truy váº¿t ngÆ°á»£c (Backtracking) Ä‘á»ƒ tÃ¬m danh sÃ¡ch cÃ¡c frame tá»‘i Æ°u
    aligned_frames = []
    t = T - 1
    j = N
    while j > 0 and t >= 0:
        if parent[t, j] == -2 or t == j - 1:
            aligned_frames.append(t)
            j -= 1
            t -= 1
        else:
            t = parent[t, j]

    aligned_frames.reverse()
    return aligned_frames, float(dp[T - 1, N])

def solve_task3(query_events, fused_candidates, keyframes_dir, dense_searcher):
    """
    Giáº£i quyáº¿t Task 3: CÄƒn chá»‰nh chuá»—i sá»± kiá»‡n theo thá»i gian (TRAKE)
    1. Nháº­n cÃ¢u mÃ´ táº£ chuá»—i sá»± kiá»‡n vÃ  mÃ£ hÃ³a tá»«ng sá»± kiá»‡n con thÃ nh vector.
    2. Chá»n video á»©ng viÃªn hÃ ng Ä‘áº§u tá»« káº¿t quáº£ Fusion.
    3. TÃ­nh toÃ¡n ma tráº­n Ä‘á»™ tÆ°Æ¡ng Ä‘á»“ng Cosine giá»¯a cÃ¡c frame vÃ  chuá»—i sá»± kiá»‡n.
    4. Ãp dá»¥ng quy hoáº¡ch Ä‘á»™ng Ä‘á»ƒ tÃ¬m vÃ  cÄƒn chá»‰nh chuá»—i frame thá»a mÃ£n thá»© tá»± thá»i gian tÄƒng dáº§n.
    """
    if not fused_candidates or not query_events:
        return {"video_id": "none", "frame_ids": []}
        
    best_candidate = fused_candidates[0]
    video_id = best_candidate["video_id"]
    
    # Äá»c file vector npy cá»§a video á»©ng viÃªn tá»‘t nháº¥t
    feature_file = glob.glob(os.path.join(dense_searcher.features_dir, f"**/{video_id}.npy"), recursive=True)
    if not feature_file:
        feature_file = glob.glob(os.path.join(dense_searcher.features_dir, f"{video_id}.npy"))
        
    if not feature_file:
        return {"video_id": video_id, "frame_ids": ["0000"] * len(query_events)}
        
    try:
        video_features = np.load(feature_file[0]) # Shape: [T, D]
    except Exception as e:
        print(f"TRAKE: Lá»—i load vector Ä‘áº·c trÆ°ng cá»§a video {video_id}: {e}")
        return {"video_id": video_id, "frame_ids": ["0000"] * len(query_events)}
        
    # MÃ£ hÃ³a cÃ¡c mÃ´ táº£ sá»± kiá»‡n con thÃ nh vector
    event_vectors = []
    for event_text in query_events:
        vector = dense_searcher.encode_text(event_text)
        event_vectors.append(vector)
    event_vectors = np.array(event_vectors) # Shape: [N, D]
    
    # TÃ­nh ma tráº­n Ä‘á»™ tÆ°Æ¡ng Ä‘á»“ng Cosine giá»¯a T frame vÃ  N sá»± kiá»‡n
    scores_matrix = np.dot(video_features, event_vectors.T) # Shape: [T, N]
    
    # Ãp dá»¥ng Quy hoáº¡ch Ä‘á»™ng Ä‘á»ƒ giáº£i bÃ i toÃ¡n cÄƒn chá»‰nh tuáº§n tá»±
    aligned_indices, path_score = align_events_dynamic_programming(scores_matrix)
    
    # Ãnh xáº¡ cÃ¡c chá»‰ sá»‘ index tá»‘i Æ°u sang frame_id thá»±c táº¿
    frame_ids = []
    for idx in aligned_indices:
        fid = get_frame_id_from_idx(keyframes_dir, video_id, idx)
        frame_ids.append(fid)
        
    return {
        "video_id": video_id,
        "frame_ids": frame_ids
    }
